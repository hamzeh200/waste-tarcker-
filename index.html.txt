    <script>
        // Data State
        const state = {
            categories: { Biodegradable: 0.0, Recyclable: 0.0, Residual: 0.0, Special: 0.0 },
            records: []
        };

        const categoryMeta = {
            Biodegradable: { bg: 'var(--bio-bg)', text: 'var(--biodegradable)', icon: 'leaf' },
            Recyclable: { bg: 'var(--recyc-bg)', text: 'var(--recyclable)', icon: 'recycle' },
            Residual: { bg: 'var(--res-bg)', text: 'var(--residual)', icon: 'trash-2' },
            Special: { bg: 'var(--spec-bg)', text: 'var(--special)', icon: 'biohazard' }
        };

        let categoryChart, trendChart;

        // Generate QR Code targeting the page URL or fallback URL
        function generateQRCode() {
            const qrContainer = document.getElementById("qrcode");
            qrContainer.innerHTML = ""; // Clear existing QR Code
            
            // Automatically uses current URL, or defaults to fallback if opened as local file
            const currentURL = (window.location.href && !window.location.href.startsWith('file://')) 
                ? window.location.href 
                : "https://example.com"; 

            new QRCode(qrContainer, {
                text: currentURL,
                width: 110,
                height: 110,
                colorDark: "#1b4332",
                colorLight: "#ffffff",
                correctLevel: QRCode.CorrectLevel.H
            });
        }

        // Login Handler
        function handleLogin(event) {
            event.preventDefault();
            const username = document.getElementById('loginUser').value.trim() || 'Resident';

            document.getElementById('displayUser').innerText = username;
            document.getElementById('login-screen').style.display = 'none';
            document.getElementById('app-screen').style.display = 'block';

            initCharts();
            renderTables();
            updateMetricsAndGoals();
            lucide.createIcons();
        }

        // Logout Handler
        function handleLogout() {
            document.getElementById('app-screen').style.display = 'none';
            document.getElementById('login-screen').style.display = 'flex';
            document.getElementById('loginUser').value = '';
            document.getElementById('loginPass').value = '';
        }

        // View Navigation Switcher
        function switchView(viewId, element) {
            document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));

            document.getElementById(viewId).classList.add('active');
            element.classList.add('active');
        }

        // Add Dynamic Waste Log Entry
        function addWasteEntry() {
            const household = document.getElementById('inputHousehold').value || 'Household';
            const type = document.getElementById('inputType').value;
            const weight = parseFloat(document.getElementById('inputWeight').value) || 0;

            if (weight <= 0) return alert('Please enter a weight value greater than 0.');

            const now = new Date();
            const dateStr = now.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });

            state.records.unshift({ date: dateStr, household, type, weight });
            state.categories[type] += weight;

            renderTables();
            updateMetricsAndGoals();
            updateCharts();

            document.getElementById('inputHousehold').value = '';
            document.getElementById('inputWeight').value = '';
        }

        // Reset Functionality
        function resetAllData() {
            if (confirm('Are you sure you want to reset all metrics and delete all logged records?')) {
                state.categories = { Biodegradable: 0.0, Recyclable: 0.0, Residual: 0.0, Special: 0.0 };
                state.records = [];

                renderTables();
                updateMetricsAndGoals();
                updateCharts();
            }
        }

        // Render Dashboard Table and Full Records Table
        function renderTables() {
            const tbodyDashboard = document.getElementById('recordsTableBody');
            const tbodyFull = document.getElementById('fullRecordsTableBody');

            tbodyDashboard.innerHTML = '';
            tbodyFull.innerHTML = '';

            if (state.records.length === 0) {
                tbodyDashboard.innerHTML = `<tr><td colspan="4" style="text-align:center; color: var(--text-muted);">No records logged yet. Use the panel above to add data.</td></tr>`;
                tbodyFull.innerHTML = `<tr><td colspan="5" style="text-align:center; color: var(--text-muted);">No submissions recorded.</td></tr>`;
                return;
            }

            state.records.slice(0, 5).forEach(rec => {
                const meta = categoryMeta[rec.type];
                tbodyDashboard.innerHTML += `
                    <tr>
                        <td>${rec.date}</td>
                        <td>${rec.household}</td>
                        <td>
                            <span class="type-badge" style="background-color: ${meta.bg}; color: ${meta.text};">
                                <i data-lucide="${meta.icon}" size="12"></i> ${rec.type}
                            </span>
                        </td>
                        <td><b>${rec.weight.toFixed(1)} kg</b></td>
                    </tr>
                `;
            });

            state.records.forEach(rec => {
                const meta = categoryMeta[rec.type];
                tbodyFull.innerHTML += `
                    <tr>
                        <td>${rec.date}</td>
                        <td>${rec.household}</td>
                        <td>
                            <span class="type-badge" style="background-color: ${meta.bg}; color: ${meta.text};">
                                <i data-lucide="${meta.icon}" size="12"></i> ${rec.type}
                            </span>
                        </td>
                        <td><b>${rec.weight.toFixed(1)} kg</b></td>
                        <td><span style="color: #2e7d32; font-weight: 600;">Logged</span></td>
                    </tr>
                `;
            });

            lucide.createIcons();
        }

        // Calculate and Update Metrics
        function updateMetricsAndGoals() {
            const total = Object.values(state.categories).reduce((a, b) => a + b, 0);

            document.getElementById('val-total').innerText = `${total.toFixed(1)} kg`;
            document.getElementById('val-bio').innerText = `${state.categories.Biodegradable.toFixed(1)} kg`;
            document.getElementById('val-recyclable').innerText = `${state.categories.Recyclable.toFixed(1)} kg`;
            document.getElementById('val-residual').innerText = `${state.categories.Residual.toFixed(1)} kg`;
            document.getElementById('val-special').innerText = `${state.categories.Special.toFixed(1)} kg`;

            const calcPct = (val) => total > 0 ? Math.round((val / total) * 100) : 0;
            document.getElementById('pct-bio').innerText = `${calcPct(state.categories.Biodegradable)}% of total`;
            document.getElementById('pct-recyclable').innerText = `${calcPct(state.categories.Recyclable)}% of total`;
            document.getElementById('pct-residual').innerText = `${calcPct(state.categories.Residual)}% of total`;
            document.getElementById('pct-special').innerText = `${calcPct(state.categories.Special)}% of total`;

            // Goal Calculations
            const targetGoal = 15;
            const progress = Math.min(Math.round((total / targetGoal) * 100), 100);

            document.getElementById('goalProgressBar').style.width = `${progress}%`;
            document.getElementById('goalRatioText').innerText = `${total.toFixed(1)} kg / ${targetGoal} kg`;
            document.getElementById('goalPercentText').innerText = `${progress}%`;

            // Update Community Progress View
            const uniqueHouseholds = new Set(state.records.map(r => r.household)).size;
            document.getElementById('comm-participating-households').innerText = `${uniqueHouseholds} / 200`;
            document.getElementById('comm-total-weight').innerText = `${total.toFixed(1)} kg`;
            document.getElementById('comm-active-rate').innerText = `${Math.round((uniqueHouseholds / 200) * 100)}%`;
        }

        // Initialize Charts
        function initCharts() {
            if (categoryChart) categoryChart.destroy();
            if (trendChart) trendChart.destroy();

            const ctxCat = document.getElementById('categoryChart').getContext('2d');
            categoryChart = new Chart(ctxCat, {
                type: 'doughnut',
                data: {
                    labels: ['Biodegradable', 'Recyclable', 'Residual', 'Special'],
                    datasets: [{
                        data: [0, 0, 0, 0],
                        backgroundColor: ['#43a047', '#1e88e5', '#757575', '#e53935']
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'right' } } }
            });

            const ctxTrend = document.getElementById('trendChart').getContext('2d');
            trendChart = new Chart(ctxTrend, {
                type: 'bar',
                data: {
                    labels: ['Week 1', 'Week 2', 'Week 3', 'Week 4', 'Current Session'],
                    datasets: [
                        { label: 'Biodegradable', data: [0, 0, 0, 0, 0], backgroundColor: '#43a047' },
                        { label: 'Recyclable', data: [0, 0, 0, 0, 0], backgroundColor: '#1e88e5' },
                        { label: 'Residual', data: [0, 0, 0, 0, 0], backgroundColor: '#757575' },
                        { label: 'Special', data: [0, 0, 0, 0, 0], backgroundColor: '#e53935' }
                    ]
                },
                options: { responsive: true, maintainAspectRatio: false, scales: { x: { stacked: true }, y: { stacked: true, beginAtZero: true } } }
            });
        }

        // Update Charts Data
        function updateCharts() {
            if (!categoryChart || !trendChart) return;
            categoryChart.data.datasets[0].data = Object.values(state.categories);
            categoryChart.update();

            trendChart.data.datasets[0].data[4] = state.categories.Biodegradable;
            trendChart.data.datasets[1].data[4] = state.categories.Recyclable;
            trendChart.data.datasets[2].data[4] = state.categories.Residual;
            trendChart.data.datasets[3].data[4] = state.categories.Special;
            trendChart.update();
        }

        // Startup Initializer
        window.addEventListener('DOMContentLoaded', () => {
            lucide.createIcons();
            generateQRCode();
        });
    </script>
